@pytest.fixture
def api():
    app = FastAPI()
    app.include_router(router, prefix='/api/meeting-rooms')
    app.add_exception_handler(AppError, app_error_handler)
    return app


@pytest.mark.asyncio
async def test_device_rotation_revocation_and_room_binding(world):
    old = await issue_device('omm_one')
    fresh = await issue_device('omm_one')
    assert (await authenticate_device(fresh, '/api/meeting-rooms/display', 'GET'))['room_id'] == 'omm_one'
    with pytest.raises(UnauthorizedError):
        await authenticate_device(old, '/api/meeting-rooms/display', 'GET')
    await revoke_device('omm_one')
    with pytest.raises(UnauthorizedError):
        await authenticate_device(fresh, '/api/meeting-rooms/display', 'GET')
    assert fresh not in str(world.values)


@pytest.mark.asyncio
@pytest.mark.parametrize('path,method', [('/api/admin/users', 'GET'), ('/api/meeting-rooms/device', 'POST'), ('/api/meeting-rooms/display', 'POST')])
async def test_display_credentials_cannot_access_other_routes(world, path, method):
    token = await issue_device('omm_one')
    request = Request({'type': 'http', 'path': path, 'method': method, 'headers': []})
    with pytest.raises(UnauthorizedError):
        await get_current_user(request, 'Bearer ' + token)


@pytest.mark.asyncio
async def test_redis_failure_denies_device_access(world):
    from redis.exceptions import ConnectionError
    world.cache.get.side_effect = ConnectionError('unavailable')
    with pytest.raises(ExternalAPIError):
        await authenticate_device('room:omm_one:' + 'a' * 43, '/api/meeting-rooms/display', 'GET')


@pytest.mark.asyncio
@pytest.mark.parametrize('method,path', [('GET', ''), ('GET', '/display'), ('POST', '/device'), ('POST', '/device/revoke')])
async def test_all_endpoints_require_auth(api, world, method, path):
    api.dependency_overrides[block_writes_in_readonly] = lambda: None
    async with HttpClient(transport=ASGITransport(app=api), base_url='http://test') as client:
        response = await client.request(method, '/api/meeting-rooms' + path, json={'room_id': 'omm_one'})
    assert response.status_code == 401


@pytest.mark.asyncio
async def test_device_display_success_room_isolation_and_validation(api, world):
    token = await issue_device('omm_one')
    async with HttpClient(transport=ASGITransport(app=api), base_url='http://test', headers={'Authorization': 'Bearer ' + token}) as client:
        response = await client.get('/api/meeting-rooms/display')
        assert response.status_code == 200
        assert response.headers['cache-control'] == 'no-store'
        assert response.json()['data']['server_time']
        assert (await client.get('/api/meeting-rooms/display?room_id=omm_two')).status_code == 403
        assert (await client.get('/api/meeting-rooms/display?room_id=invalid')).status_code == 422


@pytest.mark.asyncio
async def test_admin_catalog_issue_revoke_and_body_validation(api, world):
    api.dependency_overrides[require_admin] = lambda: {'role': 'super_admin'}
    api.dependency_overrides[block_writes_in_readonly] = lambda: None
    async with HttpClient(transport=ASGITransport(app=api), base_url='http://test') as client:
        assert (await client.get('/api/meeting-rooms')).status_code == 200
        issued = await client.post('/api/meeting-rooms/device', json={'room_id': 'omm_one'})
        assert issued.status_code == 200
        assert issued.headers['cache-control'] == 'no-store'
        assert (await client.post('/api/meeting-rooms/device/revoke', json={'room_id': 'omm_one'})).status_code == 200
        for path in ['/device', '/device/revoke']:
            assert (await client.post('/api/meeting-rooms' + path, json={'room_id': '../bad'})).status_code == 422
        assert (await client.post('/api/meeting-rooms/device', json={'room_id': 'omm_missing'})).status_code == 404
